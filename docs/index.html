Perusall Article Downloader

Instructions
  1. Open your Perusall article in Chrome/Firefox
  2. Press F12 to open Developer Tools
  3. Click on Console tab
  4. Copy and paste the code below into the console
  5. Wait for the PDF to be generated and downloaded
JavaScript Code 
void (async function () {
    // Perusall Article Downloader
    // Builds the PDF without jsPDF. Perusall's Content-Security-Policy blocks
    // cdn.jsdelivr.net, so the library can't be loaded. The page images are
    // JPEGs, so they are embedded straight into a PDF file.

    if (window.downloaderRunning) {
        console.log('Downloader already running...');
        return;
    }
    window.downloaderRunning = true;

    const PAGE_WIDTH = 595.28;   // A4 in points
    const PAGE_HEIGHT = 841.89;
    const CHUNKS_PER_PAGE = 6;

    const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

    try {
        // PDF_WRITER_START
        const dataUrlToBytes = (dataUrl) => {
            const bin = atob(dataUrl.split(',')[1]);
            const bytes = new Uint8Array(bin.length);
            for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
            return bytes;
        };

        // pages: array of pages; each page is an array of 6 slots holding
        // { bytes, width, height } (JPEG data) or null for a missing chunk.
        const buildPdf = (pages, pageWidth, pageHeight) => {
            const enc = new TextEncoder();
            const parts = [];
            const offsets = [];
            let offset = 0;

            const push = (data) => {
                const bytes = typeof data === 'string' ? enc.encode(data) : data;
                parts.push(bytes);
                offset += bytes.length;
            };

            const writeObj = (id, dict, stream) => {
                offsets[id] = offset;
                if (stream) {
                    push(`${id} 0 obj\n<< ${dict} /Length ${stream.length} >>\nstream\n`);
                    push(stream);
                    push('\nendstream\nendobj\n');
                } else {
                    push(`${id} 0 obj\n<< ${dict} >>\nendobj\n`);
                }
            };

            // Object numbers: 1 = catalog, 2 = page tree, then one id per image,
            // then a (page, content) pair per page.
            let nextId = 3;
            const imageIds = pages.map((chunks) => chunks.map((c) => (c ? nextId++ : null)));
            const pageIds = pages.map(() => [nextId++, nextId++]);
            const lastId = nextId - 1;

            // %PDF-1.4 header with binary marker bytes
            push(new Uint8Array([0x25, 0x50, 0x44, 0x46, 0x2d, 0x31, 0x2e, 0x34, 0x0a,
                                 0x25, 0xe2, 0xe3, 0xcf, 0xd3, 0x0a]));
            writeObj(1, '/Type /Catalog /Pages 2 0 R');
            writeObj(2, `/Type /Pages /Count ${pages.length} /Kids [${pageIds.map(([p]) => `${p} 0 R`).join(' ')}]`);

            pages.forEach((chunks, pi) => {
                chunks.forEach((chunk, ci) => {
                    if (!chunk) return;
                    writeObj(imageIds[pi][ci],
                        `/Type /XObject /Subtype /Image /Width ${chunk.width} /Height ${chunk.height} ` +
                        `/ColorSpace /DeviceRGB /BitsPerComponent 8 /Filter /DCTDecode`,
                        chunk.bytes);
                });
            });

            const chunkHeight = pageHeight / CHUNKS_PER_PAGE;
            pages.forEach((chunks, pi) => {
                const [pageId, contentId] = pageIds[pi];
                const xobjects = [];
                let ops = '';
                chunks.forEach((chunk, ci) => {
                    if (!chunk) return;
                    const name = `Im${ci}`;
                    xobjects.push(`/${name} ${imageIds[pi][ci]} 0 R`);
                    const y = pageHeight - (ci + 1) * chunkHeight;   // PDF origin is bottom-left
                    ops += `q ${pageWidth} 0 0 ${chunkHeight} 0 ${y} cm /${name} Do Q\n`;
                });
                writeObj(contentId, '', enc.encode(ops));
                writeObj(pageId,
                    `/Type /Page /Parent 2 0 R /MediaBox [0 0 ${pageWidth} ${pageHeight}] ` +
                    `/Resources << /XObject << ${xobjects.join(' ')} >> >> /Contents ${contentId} 0 R`);
            });

            const xrefOffset = offset;
            let xref = `xref\n0 ${lastId + 1}\n0000000000 65535 f \n`;
            for (let id = 1; id <= lastId; id++) {
                xref += `${String(offsets[id]).padStart(10, '0')} 00000 n \n`;
            }
            push(xref);
            push(`trailer\n<< /Size ${lastId + 1} /Root 1 0 R >>\nstartxref\n${xrefOffset}\n%%EOF\n`);

            return new Blob(parts, { type: 'application/pdf' });
        };
        // PDF_WRITER_END

        // Scroll through the article and collect the page chunk images.
        const collectChunkImages = async () => {
            window.scrollTo(0, 0);
            await sleep(2000);

            const imageMap = new Map();   // src -> Image, so each URL is kept once
            let maxSeen = 0;
            let stableTicks = 0;

            for (let tick = 0; tick < 150; tick++) {   // safety cap, about 5 minutes
                await sleep(2000);
                const imgs = document.querySelectorAll('img.chunk');

                imgs.forEach((img, index) => {
                    if (!img.complete || img.naturalWidth === 0 || imageMap.has(img.src)) return;
                    const image = new Image();
                    image.crossOrigin = 'anonymous';
                    image.src = img.src;
                    image.dataset.pageNumber = String(Math.floor(index / CHUNKS_PER_PAGE) + 1);
                    image.dataset.position = String(index % CHUNKS_PER_PAGE);
                    imageMap.set(img.src, image);
                });
                if (imgs.length > 0) imgs[imgs.length - 1].scrollIntoView();

                if (imgs.length > maxSeen) {
                    maxSeen = imgs.length;
                    stableTicks = 0;
                } else {
                    stableTicks++;
                }
                if (maxSeen > 0 && stableTicks > 3) break;
            }

            if (maxSeen === 0) {
                throw new Error('No page images (img.chunk) found. Let the article finish loading and try again.');
            }
            return {
                pages: Math.ceil(maxSeen / CHUNKS_PER_PAGE),
                images: Array.from(imageMap.values()),
            };
        };

        // Draw one image to a canvas and return its JPEG bytes.
        const toJpeg = (img) => new Promise((resolve, reject) => {
            const convert = () => {
                try {
                    const canvas = document.createElement('canvas');
                    canvas.width = img.naturalWidth;
                    canvas.height = img.naturalHeight;
                    canvas.getContext('2d').drawImage(img, 0, 0);
                    resolve({
                        bytes: dataUrlToBytes(canvas.toDataURL('image/jpeg', 0.75)),
                        width: canvas.width,
                        height: canvas.height,
                    });
                } catch (err) {
                    reject(err);
                }
            };
            if (img.complete && img.naturalWidth > 0) {
                convert();
            } else if (img.complete) {
                reject(new Error('Image failed to load: ' + img.src));
            } else {
                img.onload = convert;
                img.onerror = () => reject(new Error('Image failed to load: ' + img.src));
            }
        });

        console.log('Collecting page images...');
        const { pages: pageCount, images } = await collectChunkImages();
        console.log(`Detected ${pageCount} page(s), ${images.length} unique image(s).`);

        const settled = await Promise.allSettled(images.map(toJpeg));
        const pageGroups = new Map();
        settled.forEach((res, i) => {
            const img = images[i];
            if (res.status === 'rejected') {
                console.warn('Skipped image:', img.src, res.reason);
                return;
            }
            const pageNum = parseInt(img.dataset.pageNumber, 10) || 1;
            const position = parseInt(img.dataset.position, 10) || 0;
            if (!pageGroups.has(pageNum)) {
                pageGroups.set(pageNum, new Array(CHUNKS_PER_PAGE).fill(null));
            }
            pageGroups.get(pageNum)[position] = res.value;
        });

        const pages = [...pageGroups.keys()].sort((a, b) => a - b).map((num) => {
            const chunks = pageGroups.get(num);
            const missing = chunks.flatMap((c, i) => (c ? [] : [i]));
            if (missing.length) {
                console.warn(`Page ${num} is missing chunk(s) at position(s): ${missing.join(', ')}`);
            }
            return chunks;
        });
        if (pages.length === 0) {
            throw new Error('No page images could be read.');
        }

        const title = (
            document.querySelector('h1.p-title')?.textContent?.trim() ||
            document.querySelector('.p-title')?.textContent?.trim() ||
            document.querySelector('h1')?.textContent?.trim() ||
            document.querySelector('title')?.textContent?.trim() ||
            'perusall_article'
        ).replace(/[^a-z0-9]/gi, '_').toLowerCase();

        console.log('Building PDF...');
        const blob = buildPdf(pages, PAGE_WIDTH, PAGE_HEIGHT);

        const url = URL.createObjectURL(blob);
        const link = document.createElement('a');
        link.href = url;
        link.download = `${title}.pdf`;
        document.body.appendChild(link);
        link.click();
        link.remove();
        setTimeout(() => URL.revokeObjectURL(url), 60000);

        console.log(`Saved ${title}.pdf with ${pages.length} page(s).`);
    } catch (err) {
        console.error('Downloader error:', err);
    } finally {
        window.downloaderRunning = false;
    }
})();
